/** * Church Member Registry - backend (Google Apps Script) * Bound to the Google Sheet: Extensions > Apps Script. * * ACCESS CONTROL (no Google login needed): * - ENTRY code: can add members only (give to data-entry volunteers) * - ADMIN code: can add, search and view members (secretary / pastors) * Codes live in Script Properties, never in the web page. */ const SHEET_NAME = 'Members'; const HEADERS = [ 'Member ID', 'Date Added', 'First Name', 'Last Name', 'Gender', 'Date of Birth', 'Phone', 'Email', 'Address', 'Marital Status', 'Occupation', 'Member Type', 'Date Joined', 'Baptized', 'Ministry', 'Fellowship Group', 'Invited By', 'Follow-up Status', 'Status', 'Notes' ]; // EDIT THESE, then run setup() once from the editor. Change them again anytime by re-running. const INITIAL_ENTRY_CODE = 'admin'; const INITIAL_ADMIN_CODE = 'admin'; /** JSON API used by the installable app (index.html on free static hosting). */ function doPost(e) { let out; try { const req = JSON.parse(e.postData.contents); const api = { login: login, addMember: addMember, listMembers: listMembers }; if (!api[req.fn]) throw new Error('Unknown request.'); out = { result: api[req.fn].apply(null, req.args || []) }; } catch (err) { out = { error: String(err.message || err) }; } return ContentService.createTextOutput(JSON.stringify(out)).setMimeType(ContentService.MimeType.JSON); } function doGet() { return HtmlService.createHtmlOutputFromFile('Index') .setTitle('Member Registry') .addMetaTag('viewport', 'width=device-width, initial-scale=1'); } /** Run once: creates the Members tab and stores the access codes. */ function setup() { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sh = ss.getSheetByName(SHEET_NAME) || ss.insertSheet(SHEET_NAME); sh.getRange(1, 1, 1, HEADERS.length).setValues([HEADERS]) .setFontWeight('bold').setBackground('#17493F').setFontColor('#FFFFFF'); sh.setFrozenRows(1); sh.getRange(2, 1, sh.getMaxRows() - 1, HEADERS.length).setNumberFormat('@'); // plain text sh.autoResizeColumns(1, HEADERS.length); PropertiesService.getScriptProperties().setProperties({ ENTRY_CODE: INITIAL_ENTRY_CODE, ADMIN_CODE: INITIAL_ADMIN_CODE }); } /* ---------- access control ---------- */ function roleFor_(code) { const cache = CacheService.getScriptCache(); const fails = Number(cache.get('fails') || 0); if (fails >= 8) throw new Error('Too many wrong codes. Try again in 10 minutes.'); const p = PropertiesService.getScriptProperties(); const c = String(code || '').trim(); let role = null; if (c && c === p.getProperty('ADMIN_CODE')) role = 'admin'; else if (c && c === p.getProperty('ENTRY_CODE')) role = 'entry'; if (!role) { cache.put('fails', String(fails + 1), 600); throw new Error('That access code is not correct.'); } return role; } function login(code) { return { role: roleFor_(code) }; } /* ---------- members ---------- */ function addMember(code, d) { roleFor_(code); const first = clean_(d.firstName), last = clean_(d.lastName); if (!first || !last) throw new Error('First and last name are required.'); const phone = clean_(d.phone); if (!phone) throw new Error('A phone number is required.'); const lock = LockService.getScriptLock(); lock.waitLock(20000); try { const sh = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME); const last_row = sh.getLastRow(); const rows = last_row > 1 ? sh.getRange(2, 1, last_row - 1, HEADERS.length).getValues() : []; const digits = s => String(s).replace(/\D/g, ''); const dup = rows.find(r => digits(r[6]) && digits(r[6]) === digits(phone)); if (dup && !d.confirmDuplicate) { return { duplicate: true, name: dup[2] + ' ' + dup[3], id: dup[0] }; } const nums = rows.map(r => parseInt(String(r[0]).replace(/\D/g, ''), 10) || 0); const id = 'M-' + String((nums.length ? Math.max.apply(null, nums) : 0) + 1).padStart(4, '0'); const isNew = d.memberType === 'New member'; const row = [ id, Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'yyyy-MM-dd HH:mm'), first, last, clean_(d.gender), clean_(d.dob), phone, clean_(d.email), clean_(d.address), clean_(d.marital), clean_(d.occupation), isNew ? 'New member' : 'Existing member', clean_(d.dateJoined), clean_(d.baptized), clean_(d.ministry), clean_(d.group), clean_(d.invitedBy), isNew ? 'Needs follow-up' : '', 'Active', clean_(d.notes) ]; const target = sh.getRange(sh.getLastRow() + 1, 1, 1, HEADERS.length); target.setNumberFormat('@').setValues([row]); return { ok: true, id: id, name: first + ' ' + last }; } finally { lock.releaseLock(); } } function listMembers(code, query) { if (roleFor_(code) !== 'admin') throw new Error('This needs the admin code.'); const sh = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME); const n = sh.getLastRow() - 1; if (n < 1) return { total: 0, rows: [] }; const rows = sh.getRange(2, 1, n, HEADERS.length).getValues().reverse(); const q = String(query || '').toLowerCase().trim(); const hit = r => !q || r.join(' ').toLowerCase().indexOf(q) > -1; const out = rows.filter(hit).slice(0, 100).map(r => ({ id: r[0], name: r[2] + ' ' + r[3], phone: r[6], type: r[11], joined: r[12], ministry: r[14], group: r[15], followUp: r[17] })); return { total: n, rows: out }; } function clean_(v) { return String(v == null ? '' : v).trim().slice(0, 500); }
